function normalizeName(n) { return String(n == null ? '' : n).trim().toLowerCase(); }

function sumReceiptsFor(materialName) {
const key = normalizeName(materialName);
return receipts.reduce((s, r) =>
normalizeName(r.materialName) === key ? s + (Number(r.quantity) || 0) : s, 0);

function sumIssuesFor(materialName) {

return issues.reduce((s, i) =>

function computeOpeningBalanceFor(materialName, desiredQuantity) {
return (Number(desiredQuantity) || 0)
- sumReceiptsFor(materialName)

/* ===================== CATEGORY KPI (Material-wise Monthly Consumption) =====================
Built LIVE from the current issues array.
- Reversed issues are removed from the array → they disappear from this card too.
- Material matching uses normalizedName so case / spacing differences are ignored.
- Month matching uses the "YYYY-MM" prefix of the date string.
- Materials with no issues this month still appear with 0.00 so the layout stays stable.
============================================================================= */
function renderCategoryKpi() {
const container = document.getElementById('categoryKpi');
if (!container) return;
container.innerHTML = '';

const now = new Date();
const ym = now.getFullYear() + '-' + String(now.getMonth() + 1).padStart(2, '0');

const map = new Map();

issues.forEach(iss => {
if (!iss || !iss.date) return;
if (String(iss.date).slice(0, 7) !== ym) return;
const key = normalizeName(iss.materialName);
if (!key) return;
const entry = map.get(key) || { displayName: iss.materialName || '—', total: 0 };
entry.total += Number(iss.quantity) || 0;
map.set(key, entry);
});

materials.forEach(m => {
const key = normalizeName(m.name);
if (!key) return;
if (!map.has(key)) map.set(key, { displayName: m.name, total: 0 });
});

if (map.size === 0) {
container.innerHTML = '<p style="color:#999; padding: 10px;">No materials available</p>';
return;
}

const rows = [...map.values()].sort((a, b) => b.total - a.total);

const palette = ['#667eea','#4299e1','#48bb78','#ed8936','#f56565','#38b2ac','#805ad5','#d69e2e'];

rows.forEach((row, i) => {
const card = document.createElement('div');
card.className = 'category-kpi';
card.style.borderLeftColor = palette[i % palette.length];
card.innerHTML = `
<div class="ck-name" title="${escapeHtml(row.displayName)}">${escapeHtml(row.displayName)}</div>
<div class="ck-value">${row.total.toFixed(2)}</div>
`;
container.appendChild(card);
});
}

function computeRecalcChanges() {
ensureOpeningBalances();
const changes = [];
materials.forEach(m => {
const rec = sumReceiptsFor(m.name);
const iss = sumIssuesFor(m.name);
const expected = (Number(m.openingBalance) || 0) + rec - iss;
const current = Number(m.quantity) || 0;
const diff = expected - current;
if (Math.abs(diff) > 0.0001) {
changes.push({
name: m.name,
oldQty: current,
newQty: expected,
diff: diff,
opening: Number(m.openingBalance) || 0,
receipts: rec,
issues: iss
});
}
});
return changes;
}

function applyQuickRange(val) {
const now = new Date();
const fmt = d => new Date(d.getTime() - d.getTimezoneOffset() * 60000).toISOString().split('T')[0];
let from = '', to = '';

switch (val) {
case 'today': from = to = fmt(now); break;
case 'yesterday': { const y = new Date(now); y.setDate(y.getDate() - 1); from = to = fmt(y); break; }
case 'last7': { const s = new Date(now); s.setDate(s.getDate() - 6); from = fmt(s); to = fmt(now); break; }
case 'last30': { const s = new Date(now); s.setDate(s.getDate() - 29); from = fmt(s); to = fmt(now); break; }
case 'thismonth': from = fmt(new Date(now.getFullYear(), now.getMonth(), 1)); to = fmt(now); break;
case 'lastmonth': from = fmt(new Date(now.getFullYear(), now.getMonth() - 1, 1));
to = fmt(new Date(now.getFullYear(), now.getMonth(), 0)); break;
case 'thisyear': from = fmt(new Date(now.getFullYear(), 0, 1)); to = fmt(now); break;
default: from = ''; to = '';
}
document.getElementById('advFrom').value = from;
document.getElementById('advTo').value = to;
}

function buildSearchRows(f) {
const rows = [];

if (f.type === 'both' || f.type === 'receipt') {
receipts.forEach(r => rows.push({
kind: 'Receipt',
date: r.date || '',
time: r.time || '',
material: r.materialName || '',
party: r.vendorName || '',
quantity: Number(r.quantity) || 0,
by: r.receivedBy || '',
docNo: r.invoiceNumber || '',
remarks: r.remarks || ''
}));
}
if (f.type === 'both' || f.type === 'issue') {
issues.forEach(i => rows.push({
kind: 'Issue',
date: i.date || '',
time: i.time || '',
material: i.materialName || '',
party: i.issuedTo || '',
quantity: Number(i.quantity) || 0,
by: i.issuedBy || '',
docNo: i.challanNo || '',
remarks: i.remarks || ''
}));
}

return rows.filter(r => {
if (f.from && r.date < f.from) return false;
if (f.to && r.date > f.to) return false;
if (f.material && !r.material.toLowerCase().includes(f.material)) return false;
if (f.person) {
const hay = (r.party + ' ' + r.by).toLowerCase();
if (!hay.includes(f.person)) return false;
}
if (f.docNo && !String(r.docNo).toLowerCase().includes(f.docNo)) return false;
if (!isNaN(f.minQty) && r.quantity < f.minQty) return false;
if (!isNaN(f.maxQty) && r.quantity > f.maxQty) return false;
if (f.keyword) {
const hay = [r.kind, r.date, r.time, r.material, r.party, r.by, r.docNo, r.remarks, r.quantity]
.join(' ').toLowerCase();
if (!hay.includes(f.keyword)) return false;
}
return true;
}).sort((a, b) => (b.date + (b.time || '')).localeCompare(a.date + (a.time || '')));
}

let text = new TextDecoder('utf-8').decode(buffer);
if (text.charCodeAt(0) === 0xFEFF) text = text.slice(1);
const trimmed = text.trimStart();
if (trimmed.startsWith('{') || trimmed.startsWith('[')) importJSONData(JSON.parse(text));
else parseCSVAndImport(text);
}
} catch (err) { showToast('Upload failed: ' + err.message, 'error'); }
finally { fileInput.value = ''; }
};
reader.readAsArrayBuffer(file);
}

}

}
const text = await res.text();
if (!text) return null;
try { return JSON.parse(text); } catch (e) { return null; }
}

function readExcelFile(callback) {
const fileInput = document.getElementById('excelSyncFile');
const file = fileInput.files[0];
if (!file) { showToast('Please choose an Excel file first', 'error'); return; }

const reader = new FileReader();
reader.onload = function(e) {
try {
const buffer = e.target.result;
const bytes = new Uint8Array(buffer);
const isZip = bytes.length > 1 && bytes[0] === 0x50 && bytes[1] === 0x4B;
const fname = file.name.toLowerCase();

if (isZip || fname.endsWith('.xlsx') || fname.endsWith('.xls')) {
if (typeof XLSX === 'undefined') { showToast('Excel parser not loaded', 'error'); return; }
const wb = XLSX.read(buffer, { type: 'array' });
const sheetRows = XLSX.utils.sheet_to_json(wb.Sheets[wb.SheetNames[0]], { defval: '' });
callback(sheetRows);
} else if (fname.endsWith('.csv')) {
let text = new TextDecoder('utf-8').decode(buffer);
if (text.charCodeAt(0) === 0xFEFF) text = text.slice(1);
callback(parseCSVToRows(text));
} else {
showToast('Please upload .xlsx, .xls or .csv file', 'error');
}
} catch (err) { showToast('Read failed: ' + err.message, 'error'); }
};
reader.onerror = function() { showToast('Failed to read the file', 'error'); };
reader.readAsArrayBuffer(file);
}

function parseCSVToRows(csvText) {
const lines = csvText.trim().split(/\r?\n/);
if (lines.length === 0) return [];
const headers = lines[0].split(',').map(h => h.trim().replace(/^"|"$/g, ''));
const out = [];
for (let i = 1; i < lines.length; i++) {
const cells = lines[i].split(',').map(c => c.trim().replace(/^"|"$/g, ''));
const row = {};
headers.forEach((h, idx) => { row[h] = cells[idx] !== undefined ? cells[idx] : ''; });
out.push(row);
}
return out;
}

const NOTIF_PRIORITIES = { normal: 'Normal', important: 'Important', critical: 'Critical' };

/* --- expiry / status helpers --- */
function notifExpiryDate(n) {
if (!n || !n.createdAt) return null;
const days = Number(n.days) || 0;
if (days <= 0) return null; /* no expiry */
return new Date(new Date(n.createdAt).getTime() + days * 86400000);
}

function isNotifActive(n) {
if (!n || n.removed) return false;
const exp = notifExpiryDate(n);
if (!exp) return true;
return exp.getTime() > Date.now();
}

function notifDaysLeft(n) {
const exp = notifExpiryDate(n);
if (!exp) return null;
return Math.max(0, Math.ceil((exp.getTime() - Date.now()) / 86400000));
}

function activeNotifications() {
return notifications
.filter(isNotifActive)
.sort((a, b) => new Date(b.createdAt) - new Date(a.createdAt));
}

function notifReadByUser(n, userId) {
return !!(n && n.readBy && n.readBy[userId]);
}

function unreadNotificationsFor(userId) {
return activeNotifications().filter(n => !notifReadByUser(n, userId));
}

if (e.target.closest && e.target.closest('.notif-wrap')) return;
p.classList.remove('open');
});

/* --- blocking gate --- */
function showNotifGate(list) {
const gate = document.getElementById('notifGate');
const wrap = document.getElementById('notifGateList');
const btn = document.getElementById('notifGateBtn');
const hint = document.getElementById('notifGateHint');
if (!gate || !wrap || !btn) return;

wrap.innerHTML = list.map(n => {
const pri = n.priority || 'normal';
const dl = notifDaysLeft(n);
return `
<label class="notif-gate-item" data-id="${escapeHtml(String(n.id))}">
<input type="checkbox" class="notif-gate-chk"
data-id="${escapeHtml(String(n.id))}"
onchange="onGateCheckChange(this)">
<div style="flex:1; min-width:0;">
<div class="ng-title">
${escapeHtml(n.title || 'Notification')}
<span class="ng-pri ${pri}">${(NOTIF_PRIORITIES[pri] || 'Normal').toUpperCase()}</span>
</div>
<div class="ng-body">${escapeHtml(n.message || '')}</div>
<div class="ng-meta">

${new Date(n.createdAt).toLocaleString()}
${dl !== null ? ` • Visible for ${dl} more day(s)` : ''}
</div>
</div>
</label>`;

btn.disabled = true;
btn.textContent = '✅ I have read — Continue';
if (hint) hint.textContent = 'Tick every checkbox to enable the Continue button.';

gate.classList.add('active');

/* keep the bell panel closed while the gate is showing */
const p = document.getElementById('notifPanel');
if (p) p.classList.remove('open');
}

function onGateCheckChange(chk) {
const item = chk.closest ? chk.closest('.notif-gate-item') : null;
if (item) item.classList.toggle('checked', chk.checked);

const all = Array.from(document.querySelectorAll('.notif-gate-chk'));
const btn = document.getElementById('notifGateBtn');
const hint = document.getElementById('notifGateHint');
const allChecked = all.length > 0 && all.every(c => c.checked);

if (btn) btn.disabled = !allChecked;
if (hint) {
const remaining = all.filter(c => !c.checked).length;
hint.textContent = allChecked
? 'All read — you may continue.'
: `${remaining} notification(s) still unchecked.`;
}
}

function acknowledgeNotifications() {
if (!currentUser) return;

const all = Array.from(document.querySelectorAll('.notif-gate-chk'));
if (!all.length || !all.every(c => c.checked)) {
showToast('Please tick every checkbox first', 'error');
return;
}

const stamp = new Date().toISOString();
all.forEach(c => {
const n = notifications.find(x => String(x.id) === String(c.dataset.id));
if (!n) return;
if (!n.readBy) n.readBy = {};
n.readBy[currentUser.userId] = stamp;
});

localStorage.setItem('notifications', JSON.stringify(notifications));
cloudDirty = true;
scheduleCloudPush();

const gate = document.getElementById('notifGate');
if (gate) gate.classList.remove('active');

renderNotifBell();
showToast('Thank you — notification(s) marked as read', 'success');
}

function checkNotificationGate() {
if (!currentUser) return;
if (isAdmin()) return; /* admins are never blocked */

const gate = document.getElementById('notifGate');
if (gate && gate.classList.contains('active')) return;

const unread = unreadNotificationsFor(currentUser.userId);
if (unread.length === 0) return;

showNotifGate(unread);
}

/* --- admin: create / remove / list --- */
function sendNotification() {
if (!isAdmin()) { showToast('Administrator access required', 'error'); return; }

const title = document.getElementById('notifTitle').value.trim();
const message = document.getElementById('notifMessage').value.trim();
const priority = document.getElementById('notifPriority').value;
const days = parseInt(document.getElementById('notifDays').value, 10);

if (!title) { showToast('Please enter a notification title', 'error'); return; }
if (!message) { showToast('Please enter the notification message', 'error'); return; }
if (isNaN(days) || days < 1 || days > 365) {
showToast('Visible days must be between 1 and 365', 'error'); return;
}

notifications.unshift({
id: 'NTF-' + Date.now() + '-' + Math.floor(Math.random() * 1000),
title,
message,
priority,
days,
createdAt: new Date().toISOString(),
createdBy: currentUser.userId,
removed: false,
readBy: {}
});

saveToLocalStorage(); /* persists + schedules a cloud push */

document.getElementById('notifTitle').value = '';
document.getElementById('notifMessage').value = '';
document.getElementById('notifPriority').value = 'normal';
document.getElementById('notifDays').value = 7;

renderNotifAdminTable();
renderNotifBell();
pushToCloud(true);
showToast('Notification sent to all users', 'success');
}

function removeNotification(id) {
if (!isAdmin()) { showToast('Administrator access required', 'error'); return; }

const n = notifications.find(x => String(x.id) === String(id));
if (!n) { showToast('Notification not found', 'error'); return; }
if (n.removed) { showToast('Already removed', 'info'); return; }

if (!secureConfirm(`Remove notification "${n.title}"?\n\nIt will immediately disappear from every user's bell and popup.`)) return;

n.removed = true;
n.removedAt = new Date().toISOString();
n.removedBy = currentUser.userId;

saveToLocalStorage();
renderNotifAdminTable();
renderNotifBell();
pushToCloud(true);
showToast('Notification removed', 'success');
}

function renderNotifAdminTable() {
const tbody = document.getElementById('notifAdminBody');
if (!tbody) return;

if (!notifications.length) {
tbody.innerHTML = `<tr><td colspan="8" style="text-align:center;padding:26px;color:#999;">No notifications created yet</td></tr>`;
return;
}

tbody.innerHTML = notifications.map(n => {
const active = isNotifActive(n);
const dl = notifDaysLeft(n);
const readers = n.readBy ? Object.keys(n.readBy).length : 0;
const pri = n.priority || 'normal';

const statusBadge = n.removed
? '<span class="badge badge-gray">Removed</span>'
: active
? '<span class="badge badge-success">Active</span>'
: '<span class="badge badge-danger">Expired</span>';

const msg = String(n.message || '');
const shortMsg = msg.length > 120 ? msg.slice(0, 120) + '…' : msg;

return `<tr>
<td style="min-width:220px;">
<strong>${escapeHtml(n.title || '-')}</strong>
<div style="font-size:11.5px;color:#718096;margin-top:4px;white-space:pre-wrap;">${escapeHtml(shortMsg)}</div>
</td>
<td><span class="ng-pri ${pri}">${(NOTIF_PRIORITIES[pri] || 'Normal').toUpperCase()}</span></td>
<td>${new Date(n.createdAt).toLocaleString()}</td>
<td>${n.days} day${Number(n.days) > 1 ? 's' : ''}</td>
<td>${dl === null ? '—' : (active ? dl + ' day(s)' : '0')}</td>
<td>${readers}</td>
<td>${statusBadge}</td>
<td>
${n.removed ? '<span style="color:#a0aec0;font-size:12px;">—</span>'
: `<button class="btn btn-danger btn-sm" onclick="removeNotification('${escapeJs(String(n.id))}')">🗑️ Remove</button>`}
</td>
</tr>`;
}).join('');
}
